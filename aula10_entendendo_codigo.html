HTML + CSS + JS conectados

Como o CRUD funciona por dentro

O mesmo codigo da pratica, agora comentado para entender como cada parte conversa com as outras.

Entenda o HTML

O HTML cria os elementos que aparecem na tela: formulario, campos, botoes, tabela e mensagem inicial.

<!-- main e a area principal da aplicacao.
     A classe "app" sera usada pelo CSS para centralizar e limitar a largura. -->
<main class="app">

  <!-- section cria um bloco visual.
       A classe "card" sera usada pelo CSS para deixar o bloco com aparencia de cartao. -->
  <section class="card">

    <!-- header organiza titulo, etiqueta e explicacao da pratica. -->
    <div class="header">
      <span class="tag">CRUD sem banco</span>
      <h1>Cadastro de Alunos</h1>
      <p>Cadastre, liste, altere e exclua alunos usando HTML, CSS e JavaScript.</p>
    </div>

    <!-- Este formulario sera escutado pelo JavaScript.
         Conexao com JS: document.getElementById("formAluno"). -->
    <form id="formAluno" class="form">

      <!-- Cada input tem um id.
           Conexao com JS: nome.value, turma.value e cidade.value pegam o que foi digitado. -->
      <input id="nome" placeholder="Nome do aluno" required>
      <input id="turma" placeholder="Turma" required>
      <input id="cidade" placeholder="Cidade" required>

      <!-- actions e uma classe para o CSS organizar os botoes lado a lado. -->
      <div class="actions">

        <!-- Este botao envia o formulario.
             Conexao com JS: btnSalvar muda de "Cadastrar" para "Salvar alteracao". -->
        <button id="btnSalvar" type="submit">Cadastrar</button>

        <!-- Comeca escondido com a classe hidden.
             Conexao com CSS: .hidden { display: none; }
             Conexao com JS: classList.remove("hidden") faz aparecer. -->
        <button id="btnCancelar" type="button" class="secondary hidden">Cancelar alteração</button>
      </div>
    </form>
  </section>

  <section class="card">
    <h2>Alunos cadastrados</h2>

    <!-- A tabela mostra os alunos cadastrados. -->
    <table>
      <thead>
        <tr>
          <th>Nome</th>
          <th>Turma</th>
          <th>Cidade</th>
          <th>Ações</th>
        </tr>
      </thead>

      <!-- Este tbody e o lugar onde o JavaScript vai colocar novas linhas.
           Conexao com JS: listaAlunos.appendChild(linha). -->
      <tbody id="listaAlunos">
        <tr class="empty-row">
          <td colspan="4">Nenhum aluno cadastrado ainda.</td>
        </tr>
      </tbody>
    </table>
  </section>
</main>
Conexao principal: o HTML entrega os pontos de contato. O CSS usa as classes, como card, form e hidden. O JavaScript usa os ids, como formAluno, nome e listaAlunos.

Entenda o JS

O JavaScript e a parte que faz a pagina reagir: cadastra, lista, altera, exclui e limpa o formulario.

// Busca no HTML o formulario com id="formAluno".
// Conexao com HTML: <form id="formAluno" class="form">
const form = document.getElementById("formAluno");

// Busca os campos do formulario.
// Cada constante guarda o elemento inteiro, nao apenas o texto digitado.
const nome = document.getElementById("nome");
const turma = document.getElementById("turma");
const cidade = document.getElementById("cidade");

// Busca a tabela onde as linhas de alunos serao colocadas.
const listaAlunos = document.getElementById("listaAlunos");

// Busca os botoes principais do formulario.
const btnSalvar = document.getElementById("btnSalvar");
const btnCancelar = document.getElementById("btnCancelar");

// Guarda qual linha esta sendo editada.
// null significa: nao estou editando nenhuma linha agora.
let linhaEmEdicao = null;

// O evento submit acontece quando o aluno clica em Cadastrar ou pressiona Enter.
form.addEventListener("submit", function(evento) {
  // Impede o formulario de recarregar a pagina.
  evento.preventDefault();

  // Se nao ha linha em edicao, o formulario cadastra um aluno novo.
  if (linhaEmEdicao === null) {
    cadastrarAluno();
  } else {
    // Se ha uma linha em edicao, o formulario salva a alteracao.
    salvarAlteracao();
  }

  // Depois de cadastrar ou alterar, limpa os campos.
  limparFormulario();
});

// Quando clicar em Cancelar alteracao, sai do modo edicao.
btnCancelar.addEventListener("click", function() {
  linhaEmEdicao = null;
  limparFormulario();
});

function cadastrarAluno() {
  // Remove a mensagem "Nenhum aluno cadastrado ainda".
  removerMensagemVazia();

  // Cria uma linha nova de tabela: <tr>.
  const linha = document.createElement("tr");

  // Preenche a linha com celulas e botoes.
  // Conexao com CSS: class="edit" e class="delete" dao cores diferentes.
  linha.innerHTML = `
    <td>${nome.value}</td>
    <td>${turma.value}</td>
    <td>${cidade.value}</td>
    <td>
      <button class="edit" onclick="alterarAluno(this)">Alterar</button>
      <button class="delete" onclick="excluirAluno(this)">Excluir</button>
    </td>
  `;

  // Coloca a linha criada dentro do tbody id="listaAlunos".
  listaAlunos.appendChild(linha);
}

function alterarAluno(botao) {
  // this, enviado no onclick, e o botao clicado.
  // parentElement sobe para o td; outro parentElement sobe para o tr.
  linhaEmEdicao = botao.parentElement.parentElement;

  // Pega o texto das celulas da tabela e coloca de volta nos inputs.
  nome.value = linhaEmEdicao.children[0].innerText;
  turma.value = linhaEmEdicao.children[1].innerText;
  cidade.value = linhaEmEdicao.children[2].innerText;

  // Muda o texto do botao principal.
  btnSalvar.innerText = "Salvar alteração";

  // Remove a classe hidden para mostrar o botao Cancelar.
  btnCancelar.classList.remove("hidden");
}

function salvarAlteracao() {
  // Atualiza a linha da tabela com os valores atuais dos inputs.
  linhaEmEdicao.children[0].innerText = nome.value;
  linhaEmEdicao.children[1].innerText = turma.value;
  linhaEmEdicao.children[2].innerText = cidade.value;

  // Sai do modo edicao.
  linhaEmEdicao = null;
}

function excluirAluno(botao) {
  // Encontra a linha do botao clicado.
  const linha = botao.parentElement.parentElement;

  // Remove a linha da tela.
  linha.remove();

  // Se a tabela ficou vazia, mostra a mensagem inicial de novo.
  if (listaAlunos.children.length === 0) {
    mostrarMensagemVazia();
  }
}

function limparFormulario() {
  // Limpa os campos do formulario.
  form.reset();

  // Volta o botao principal ao estado inicial.
  btnSalvar.innerText = "Cadastrar";

  // Esconde novamente o botao Cancelar.
  btnCancelar.classList.add("hidden");
}

function removerMensagemVazia() {
  // Procura a linha inicial de aviso.
  const linhaVazia = document.querySelector(".empty-row");

  // Se encontrou, remove.
  if (linhaVazia !== null) {
    linhaVazia.remove();
  }
}

function mostrarMensagemVazia() {
  // Recria a linha de aviso quando nao ha alunos na tabela.
  listaAlunos.innerHTML = `
    <tr class="empty-row">
      <td colspan="4">Nenhum aluno cadastrado ainda.</td>
    </tr>
  `;
}
Conexao principal: o JavaScript usa ids para encontrar elementos, usa value para ler os campos, cria HTML novo com innerHTML e coloca esse HTML dentro da tabela com appendChild.

Mapa CRUD da pratica

CRUDFuncao no JSO que acontece na tela
CreatecadastrarAluno()Cria uma nova linha com nome, turma, cidade e botoes.
ReadlistaAlunos.appendChild(linha)Mostra o aluno cadastrado dentro da tabela.
UpdatealterarAluno() e salvarAlteracao()Leva os dados para o formulario e depois atualiza a linha.
DeleteexcluirAluno()Remove a linha do aluno da tabela.
